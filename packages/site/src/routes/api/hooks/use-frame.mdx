# useFrame

Registers a callback that will be called before every frame is rendered, useful for animations and updates.

> **60fps Animations**: useFrame runs at 60fps by default, perfect for smooth animations and real-time updates.

> **Performance**: Use refs instead of signals for high-frequency updates to avoid unnecessary reactivity overhead.

## Parameters Reference

| Parameter | Type | Description |
|-----------|------|-------------|
| **callback** | `Function` | Called each frame with context, delta, and optional XR frame |
| **context** | `Context` | Three.js context (same as `useThree()`) |
| **delta** | `number` | Time elapsed since last frame in seconds |
| **frame** | `XRFrame?` | Optional XRFrame for WebXR sessions |
| **priority** | `number` | Execution order (lower = earlier, default: 0) |
| **stage** | `"before" \| "after"` | Run before or after rendering (default: "before") |

## TypeScript Interface

```tsx
useFrame(
  callback: (context: Context, delta: number, frame?: XRFrame) => void,
  options?: {
    priority?: number
    stage?: "before" | "after"
  }
)
```

## Animation Patterns

```tsx tab title="Basic Animation"
// Simple rotation using refs for performance
function RotatingCube() {
  let mesh: THREE.Mesh = null!

  useFrame((context, delta) => {
    mesh.rotation.x += delta
    mesh.rotation.y += delta * 0.5
  })

  return (
    <T.Mesh ref={mesh}>
      <T.BoxGeometry />
      <T.MeshStandardMaterial color="orange" />
    </T.Mesh>
  )
}
```

```tsx tab title="Priority System"
// Control execution order with priority
function PriorityExample() {
  let mesh: THREE.Mesh = null!
  
  // Physics update (runs first)
  useFrame(
    (context, delta) => {
      mesh.position.x = Math.sin(context.clock.elapsedTime) * 2
    },
    { priority: -1 }
  )

  // Visual feedback (runs after physics)
  useFrame(
    (context, delta) => {
      console.log("Position:", mesh.position.x)
    },
    { priority: 10 }
  )
  
  return <T.Mesh ref={mesh}>...</T.Mesh>
}
```

```tsx tab title="Render Stages"
// Control when your callback runs relative to rendering
function StageExample() {
  let mesh: THREE.Mesh = null!
  
  // Update animations before render (default)
  useFrame(
    (context, delta) => {
      mesh.rotation.y += delta
    },
    { stage: "before" }
  )

  // Analytics or cleanup after render
  useFrame(
    (context, delta) => {
      console.info("Frame time:", delta * 1000, "ms")
    },
    { stage: "after" }
  )
  
  return <T.Mesh ref={mesh}>...</T.Mesh>
}
```

## Advanced Example

```tsx
function ComplexAnimation() {
  let mesh: THREE.Mesh = null!
  const [speed, setSpeed] = createSignal(1)
  const [paused, setPaused] = createSignal(false)

  useFrame((context, delta) => {
    if (paused()) return
    
    const time = context.clock.elapsedTime * speed()
    
    // Complex animations
    mesh.position.x = Math.sin(time) * 2
    mesh.position.y = Math.cos(time * 2) * 1
    mesh.rotation.x = time * 0.5
    mesh.rotation.y = time * 0.3
    
    // Scale pulsing
    const scale = 1 + Math.sin(time * 3) * 0.1
    mesh.scale.setScalar(scale)
  })

  return (
    <T.Mesh ref={mesh}>
      <T.IcosahedronGeometry args={[1, 2]} />
      <T.MeshPhysicalMaterial 
        color="cyan" 
        metalness={0.8} 
        roughness={0.2} 
      />
    </T.Mesh>
  )
}
```
