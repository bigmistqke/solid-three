# Resource

Wrapper component around 'useLoader' hook for declarative resource loading.

> **Declarative Loading**: Use Resource for simpler, declarative resource loading directly in JSX. Use useLoader when you need more control.

> **Automatic Caching**: Resources are cached by default for better performance. Disable only for dynamic content.

## Props Reference

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| **loader** | `LoaderConstructor` | - | Three.js loader class (TextureLoader, GLTFLoader, etc.) |
| **url** | `string \| string[]` | - | URL(s) to load (array for cube textures) |
| **children** | `(resource: T) => JSX.Element` | - | Optional render function for complex resources |
| **cache** | `boolean` | `true` | Enable/disable caching |
| **attach** | `string` | - | Property to attach the resource to on parent |
| **...props** | `any` | - | Additional props passed to loaded resource |

## Loading Patterns

```tsx tab title="Simple Textures"
// Direct attachment to material properties
<T.MeshStandardMaterial>
  <Resource loader={TextureLoader} url="diffuse.jpg" attach="map" />
  <Resource loader={TextureLoader} url="normal.jpg" attach="normalMap" />
  <Resource loader={TextureLoader} url="roughness.jpg" attach="roughnessMap" />
</T.MeshStandardMaterial>
```

```tsx tab title="Cube Textures"
// Environment maps and skyboxes
<T.Scene>
  <Resource 
    loader={CubeTextureLoader} 
    url={[
      'px.png', 'nx.png',
      'py.png', 'ny.png',
      'pz.png', 'nz.png'
    ]} 
    attach="background"
  />
</T.Scene>
```

```tsx tab title="Complex Models"
// GLTF models with render functions
<Resource loader={GLTFLoader} url="model.gltf">
  {(gltf) => (
    <Entity 
      from={gltf.scene} 
      scale={2}
      traverse={(child) => {
        if (child.isMesh) {
          child.castShadow = true
        }
      }}
    />
  )}
</Resource>
```

```tsx tab title="Dynamic Resources"
// Disable caching for frequently changing content
<Resource 
  loader={TextureLoader} 
  url={dynamicTextureUrl()} 
  cache={false}
  attach="map"
/>
```

## Advanced Examples

### Loading Environment Maps

```tsx
function Scene() {
  return (
    <T.Scene>
      <Resource 
        loader={CubeTextureLoader}
        url={[
          '/env/px.jpg', '/env/nx.jpg',
          '/env/py.jpg', '/env/ny.jpg',
          '/env/pz.jpg', '/env/nz.jpg'
        ]}
        attach="background"
      />
      <T.Mesh>
        <T.SphereGeometry args={[1, 32, 32]} />
        <T.MeshStandardMaterial metalness={1} roughness={0}>
          <Resource 
            loader={CubeTextureLoader}
            url={[
              '/env/px.jpg', '/env/nx.jpg',
              '/env/py.jpg', '/env/ny.jpg',
              '/env/pz.jpg', '/env/nz.jpg'
            ]}
            attach="envMap"
          />
        </T.MeshStandardMaterial>
      </T.Mesh>
    </T.Scene>
  )
}
```

### Complex Model Loading

```tsx
function Model() {
  return (
    <Resource loader={GLTFLoader} url="/models/character.glb">
      {(gltf) => {
        // Process animations
        const animations = gltf.animations
        const mixer = new THREE.AnimationMixer(gltf.scene)
        
        // Play first animation
        if (animations.length > 0) {
          mixer.clipAction(animations[0]).play()
        }
        
        // Update mixer in animation loop
        useFrame((_, delta) => mixer.update(delta))
        
        return (
          <Entity 
            from={gltf.scene} 
            scale={0.01}
            castShadow
            receiveShadow
          />
        )
      }}
    </Resource>
  )
}
```

### Multiple Textures with Different Settings

```tsx
function TexturedMesh() {
  return (
    <T.Mesh>
      <T.PlaneGeometry args={[5, 5]} />
      <T.MeshStandardMaterial>
        <Resource 
          loader={TextureLoader} 
          url="wood-diffuse.jpg" 
          attach="map"
          wrapS={THREE.RepeatWrapping}
          wrapT={THREE.RepeatWrapping}
          repeat={[2, 2]}
        />
        <Resource 
          loader={TextureLoader} 
          url="wood-normal.jpg" 
          attach="normalMap"
          normalScale={[0.5, 0.5]}
        />
        <Resource 
          loader={TextureLoader} 
          url="wood-ao.jpg" 
          attach="aoMap"
          aoMapIntensity={0.5}
        />
      </T.MeshStandardMaterial>
    </T.Mesh>
  )
}
```

## Integration with Suspense

Resource components work with Solid's Suspense boundaries:

```tsx
import { Suspense } from "solid-js"

function App() {
  return (
    <Canvas>
      <Suspense fallback={<LoadingSpinner />}>
        <Resource loader={GLTFLoader} url="heavy-model.glb">
          {(gltf) => <Entity from={gltf.scene} />}
        </Resource>
      </Suspense>
    </Canvas>
  )
}
```
